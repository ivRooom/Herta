cmake_minimum_required(VERSION 3.20)
project(herta_whispercpp LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_COMMON_FFMPEG OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)

if(NOT EXISTS "/opt/whisper.cpp/include/whisper.h")
  message(FATAL_ERROR "A pinned whisper.cpp source tree must be supplied by the builder")
endif()
add_subdirectory("/opt/whisper.cpp" "${CMAKE_BINARY_DIR}/whisper-upstream")
add_executable(herta-whisper-cpp main.cpp)
target_link_libraries(herta-whisper-cpp PRIVATE whisper)
target_link_options(herta-whisper-cpp PRIVATE -static)
